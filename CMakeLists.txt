cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(StorageRecorder VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio C++ tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(RECORDER_ASAN "Enable user-mode AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(RECORDER_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(StorageRecorder SDK/StorageRecorder.cpp)
target_link_libraries(StorageRecorder PUBLIC setupapi cfgmgr32)
add_executable(StorageRecorderApp WIN32 App/Main.cpp)
target_link_libraries(StorageRecorderApp PRIVATE StorageRecorder user32 gdi32 comctl32 comdlg32 shell32 gdiplus ole32)
add_executable(StorageRecorderCLI App/Console.cpp)
target_link_libraries(StorageRecorderCLI PRIVATE StorageRecorder)
add_executable(StorageRecorderExample Examples/record.cpp)
target_link_libraries(StorageRecorderExample PRIVATE StorageRecorder)
add_executable(StorageRecorderTests Tests/Protocol.cpp)
target_link_libraries(StorageRecorderTests PRIVATE StorageRecorder)
enable_testing()
add_test(NAME OfflineProtocolAndDuration COMMAND StorageRecorderTests)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft-signed INF/CAT/SYS package")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(name TitanStorageFlightRecorder.inf TitanStorageFlightRecorder.cat TitanStorageFlightRecorder.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete opaque package: ${name}")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanStorageFlightRecorder.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanStorageFlightRecorder.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanStorageFlightRecorder.sys\"\n")
endif()
add_library(SetupCore Installer/Setup.cpp)
target_link_libraries(SetupCore PUBLIC StorageRecorder setupapi newdev wintrust crypt32 advapi32 ole32)
add_executable(StorageRecorderInstallerTests Tests/Installer.cpp)
target_link_libraries(StorageRecorderInstallerTests PRIVATE SetupCore)
add_test(NAME InstallerIdentityAndSignatureRefusal COMMAND StorageRecorderInstallerTests)
add_executable(StorageRecorderSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(StorageRecorderSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(target StorageRecorderSetup StorageRecorderSetupConsole)
 target_link_libraries(${target} PRIVATE SetupCore user32 gdi32 comctl32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
foreach(target StorageRecorderApp StorageRecorderCLI StorageRecorderExample StorageRecorderTests StorageRecorderSetup StorageRecorderSetupConsole StorageRecorderInstallerTests)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
